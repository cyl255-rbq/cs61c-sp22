、完美的“函数调用开销”与流水线打断
在C语言中，调用一个函数并不只是“跳转代码"那么简单。每一次调用pow_matrix，CPU都需要
做以下事情：
将参数（result，temp，1等）压入寄存器或栈中。
2保存当前函数的上下文（保护现场）。
3执行跳转指令（cal）。
进入新函数后，分配新的栈帧，执行条件判断（if（pow ==θ）→elseif（pow==1））。
5退出时恢复现场并返回（ret）。
在版本1中，这些重复的矩阵数据移动全部被展开（Flattened）在当前的while循环内，CPU的
分支预测器（BranchPredictor）可以极高效率地预判流水线。而写成递归调用后，CPU必须频繁处
理函数调用链，导致流水线被打断。

这里因为是while，循环里一直调用，所以调用的函数的开销会非常影响总体性能
int pow_matrix(matrix *result, matrix *mat, int pow) {
    // Task 1.6 TODO
    int n = mat->cols * mat->rows;
    if (pow == 0)
    {
        fill_matrix(result, 0);
        for (int i = 0; i < mat->cols; i++)
        {
            set(result, i, i, 1);
        }
    } else if (pow == 1)
    {
        #pragma omp parallel for
        for (int i = 0; i < n / 16 * 16; i += 16)
        {
            _mm256_storeu_pd (result->data + i + 0, _mm256_loadu_pd (mat->data + i + 0));
            _mm256_storeu_pd (result->data + i + 4, _mm256_loadu_pd (mat->data + i + 4));
            _mm256_storeu_pd (result->data + i + 8, _mm256_loadu_pd (mat->data + i + 8));
            _mm256_storeu_pd (result->data + i + 12, _mm256_loadu_pd (mat->data + i + 12));
        }
        for (int i = n / 16 * 16;  i < n; i++)
        {
            result->data[i] = mat->data[i];
        }
    } else {
        matrix *temp = NULL;
        matrix *x = NULL;
        allocate_matrix(&temp, mat->rows, mat->cols);
        allocate_matrix(&x, mat->rows, mat->cols);
        // pow_matrix(result, mat, 0);
        fill_matrix(result, 0);
        for (int i = 0; i < mat->cols; i++)
        {
            set(result, i, i, 1);
        }
        // pow_matrix(x, mat, 1);
        #pragma omp parallel for
        for (int i = 0; i < n / 16 * 16; i += 16)
        {
            _mm256_storeu_pd (x->data + i + 0, _mm256_loadu_pd (mat->data + i + 0));
            _mm256_storeu_pd (x->data + i + 4, _mm256_loadu_pd (mat->data + i + 4));
            _mm256_storeu_pd (x->data + i + 8, _mm256_loadu_pd (mat->data + i + 8));
            _mm256_storeu_pd (x->data + i + 12, _mm256_loadu_pd (mat->data + i + 12));
        }
        for (int i = n / 16 * 16;  i < n; i++)
        {
            x->data[i] = mat->data[i];
        }
        while (pow > 0)
        {
            if (pow % 2 == 1)
            {
                mul_matrix(temp, result, x);
                // pow_matrix(result, temp, 1);
                #pragma omp parallel for
                for (int i = 0; i < n / 16 * 16; i += 16)
                {
                    _mm256_storeu_pd (result->data + i + 0, _mm256_loadu_pd (temp->data + i + 0));
                    _mm256_storeu_pd (result->data + i + 4, _mm256_loadu_pd (temp->data + i + 4));
                    _mm256_storeu_pd (result->data + i + 8, _mm256_loadu_pd (temp->data + i + 8));
                    _mm256_storeu_pd (result->data + i + 12, _mm256_loadu_pd (temp->data + i + 12));
                }
                for (int i = n / 16 * 16;  i < n; i++)
                {
                    result->data[i] = temp->data[i];
                }
            }
            if (pow == 1)
            {
                break;
            }
            mul_matrix(temp, x, x);
            // pow_matrix(x, temp, 1);
            #pragma omp parallel for
            for (int i = 0; i < n / 16 * 16; i += 16)
            {
                _mm256_storeu_pd (x->data + i + 0, _mm256_loadu_pd (temp->data + i + 0));
                _mm256_storeu_pd (x->data + i + 4, _mm256_loadu_pd (temp->data + i + 4));
                _mm256_storeu_pd (x->data + i + 8, _mm256_loadu_pd (temp->data + i + 8));
                _mm256_storeu_pd (x->data + i + 12, _mm256_loadu_pd (temp->data + i + 12));
            }
            for (int i = n / 16 * 16;  i < n; i++)
            {
                x->data[i] = temp->data[i];
            }
            pow /= 2;
        }
        deallocate_matrix(temp);
        deallocate_matrix(x);
    }
    return 0;
}
